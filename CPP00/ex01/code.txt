PhoneBook::PhoneBook() {
    this->_index = 0;
    this->_total = 0;
}

std::string formatColumn(std::string str) {
    if (str.length() > 10) {
        return str.substr(0, 9) + ".";
    }
    return str;
}

void PhoneBook::searchContacts() const {
    if (this->_total == 0) {
        std::cout << "The phonebook is currently empty." << std::endl;
        return;
    }

    // 1. Print the Header Table
    std::cout << "---------------------------------------------" << std::endl;
    std::cout << "|" << std::setw(10) << "Index" << "|"
              << std::setw(10) << "First Name" << "|"
              << std::setw(10) << "Last Name" << "|"
              << std::setw(10) << "Nickname" << "|" << std::endl;
    std::cout << "---------------------------------------------" << std::endl;

    // 2. Loop through and print all saved contacts
    for (int i = 0; i < this->_total; i++) {
        std::cout << "|" << std::setw(10) << i << "|"
                  << std::setw(10) << formatColumn(this->_contacts[i].getFirstName()) << "|"
                  << std::setw(10) << formatColumn(this->_contacts[i].getLastName()) << "|"
                  << std::setw(10) << formatColumn(this->_contacts[i].getNickname()) << "|" 
                  << std::endl;
    }
    std::cout << "---------------------------------------------" << std::endl;

    // 3. Ask for the specific index
    std::string input;
    std::cout << "Enter the index of the contact to view: ";
    
    if (!std::getline(std::cin, input)) {
        return; // Handle EOF gracefully
    }

    // 4. Input Validation
    // Check if input is a single digit. 
    // We use basic string checks instead of relying purely on atoi
    if (input.length() == 1 && input[0] >= '0' && input[0] <= '7') {
        int target = input[0] - '0'; // Convert char to int
        
        if (target < this->_total) {
            this->_contacts[target].displayFullContact();
        } else {
            std::cout << "Error: Contact at index " << target << " does not exist yet." << std::endl;
        }
    } else {
        std::cout << "Error: Invalid index. Please enter a number between 0 and " << (this->_total - 1) << "." << std::endl;
    }
}

PhoneBook::~PhoneBook() {}

void PhoneBook::addContact() {
    // Determine where this contact will be saved using modulo
    int targetIndex = this->_index % 8;

    std::cout << "\n--- Creating New Contact ---" << std::endl;
    
    // Call the Contact's input method. If it returns true, commit the save.
    if (this->_contacts[targetIndex].setInformation()) {
        this->_index++;
        
        // Keep track of the total for the SEARCH display (caps at 8)
        if (this->_total < 8) {
            this->_total++;
        }
        std::cout << "Success: Contact saved!" << std::endl;
    } else {
        // If it returned false, EOF was hit. The main loop will handle the exit.
        std::cout << "\nOperation aborted." << std::endl;
    }

// A private helper function (add this to your Contact.hpp under private:)
// std::string _getInput(std::string prompt) const;
std::string Contact::_getInput(std::string prompt) const {
    std::string input = "";
    
    while (input.empty()) {
        std::cout << prompt;
        if (!std::getline(std::cin, input)) {
            return ""; // EOF detected, return empty to trigger abort
        }
        if (input.empty()) {
            std::cout << "Error: Field cannot be empty. Please try again." << std::endl;
        }
    }
    return input;
}

bool Contact::setInformation() {
    this->_firstName = _getInput("Enter First Name: ");
    if (this->_firstName.empty()) return false;

    this->_lastName = _getInput("Enter Last Name: ");
    if (this->_lastName.empty()) return false;

    this->_nickname = _getInput("Enter Nickname: ");
    if (this->_nickname.empty()) return false;

    this->_phoneNumber = _getInput("Enter Phone Number: ");
    if (this->_phoneNumber.empty()) return false;

    this->_darkestSecret = _getInput("Enter Darkest Secret: ");
    if (this->_darkestSecret.empty()) return false;

    return true; // All fields successfully filled
